# 自动售卖：物品识别与交易决策

> 背包满了不能一直攒着，得**自动卖掉不值钱的**。这一篇把"识别物品价值 → 决定卖哪些 → 走到商店 → 批量出售"整条链路做出来。全程在自己/开源游戏的商店系统里实现。

## 一、交易决策：卖什么、留什么

```python
PRICES = {
    "金币": 10,
    "药水": 20,
    "装备": 50,
    "神器": 500,
}

# 背包满了，先卖低价值的
def decide_sell(inventory, keep_these=("神器",)):
    to_sell = []
    for slot in inventory.items:
        name = slot["name"]
        if name not in keep_these:      # 不保留的都卖
            to_sell.append(slot)
    return to_sell
```

## 三、界面交互：打开商店 → 出售

```python
def open_shop(hands, vision):
    hands.press("n")                 # 打开商店面板（沙盒定义）
    time.sleep(0.3)
    if not vision.find("shop_window"):
        return False                  # 没打开就重试
    return True

def sell_items(hands, vision, items):
    for slot in items:
        # 找到背包里的物品图标 → 点击 → 点"出售"
        pos = vision.find_item_icon(slot["name"])
        if pos:
            hands.click(pos)
            sell_btn = vision.find_template("btn_sell.png")
            if sell_btn:
                hands.click(sell_btn)
            time.sleep(0.1)
```

## 四、完整售卖流程（含容错）

```python
def sell_cycle(player, inventory, shop_pos, path_finder,
               vision, hands):
    if not inventory.is_full():
        return "NOT_FULL"

    # 1. 去商店
    go_to_shop(player, shop_pos, path_finder)
    if not open_shop(vision, hands):
        return "SHOP_FAIL"           # 重试/报错

    # 2. 决策并卖出
    to_sell = decide_sell(inventory)
    sold = sell_items(hands, vision, to_sell)
    for s in to_sell:
        inventory.remove(s)

    # 3. 回挂机点
    close_shop(hands)
    go_back_to_farm(player, farm_pos, path_finder)
    return f"SOLD {sold} items"
```

def market_price(name, base, wave):
    """简单价格波动：让决策更有趣"""
    return base * (1 + 0.2 * math.sin(wave + hash(name) % 10))
