```python
def classify_drop(region_img):
    """按颜色把掉落分类"""
    hsv = cv2.cvtColor(region_img, cv2.COLOR_BGR2HSV)
    if gold(hsv):      return "金币"
```

## 二、自动拾取

```python
def auto_loot(player, drops, hands):
    """优先拾取最近的金币/装备"""
    nearest = None
    best_dist = float("inf")
    for d in drops:
        ddist = math.hypot(d.x - player.x, d.y - player.y)
        if ddist < best_dist:
            best_dist = ddist
            nearest = d

    if nearest and best_dist < PICK_RADIUS:
        hands.press("f")          # 拾取
        drops.remove(nearest)
        return True
    return False
```

## 三、背包数据结构

```python
class Inventory:
    def __init__(self, cap=10):
        self.items = []     # [{"name": "sword", "count": 1}, ...]
        self.cap = cap

    def add(self, item):
        for slot in self.items:
            if slot["name"] == item["name"]:
                slot["count"] += 1
                return True
        if len(self.items) < self.cap:
            self.items.append({"name": item["name"], "count": 1})
            return True
        return False          # 满了

    def is_full(self):
        return sum(s["count"] for s in self.items) >= self.cap

    def value(self, prices):
        return sum(prices.get(s["name"], 0) * s["count"]
                   for s in self.items)
```

## 四、背包优先级排序

```python
PRICE = {"金币": 10, "装备": 50, "药水": 20}

def sort_inventory(inv):
    # 价值从高到低排序
    inv.items.sort(key=lambda s: PRICE.get(s["name"], 0),
                   reverse=True)
```

> 背包满了：优先丢弃"最不值"的，或直接回城卖。

## 五、背包满了怎么办 → 回城售卖

```python
def handle_full_inventory(player, inv, actions, town_pos):
    if inv.is_full():
        actions.move_to(town_pos)     # 走回城
        actions.open_shop()
        for slot in inv.items:
            actions.sell(slot["name"], slot["count"])
        inv.items.clear()
        actions.move_back()           # 回挂机点
```

## 六、拾取/背包完整状态

```
战斗 → 拾取(捡最近的) → 背包检查
                │
                ├─ 没满 → 继续打怪
                └─ 满了 → 回城 → 卖 → 返回挂机点
```

```python
def loot_and_manage(player, drops, inv, actions):
    if not drops and inv.is_full():
        actions.goto_town_and_sell(inv)
        return "SELLING"
    if auto_pickup(player, drops):
```
